function getColor(num) {
  if (num === 0) return 'green';
  const reds = [1,3,5,7,9,12,14,16,18,19,21,23,25,27,30,32,34,36];
  return reds.includes(num) ? 'red' : 'black';
}

function evaluateSpin(result) {
  let winnings = 0;
  if (bets[result]) winnings += bets[result] * 35;

  if (bets['red'] && getColor(result) === 'red') winnings += bets['red'] * 1;
  if (bets['black'] && getColor(result) === 'black') winnings += bets['black'] * 1;
  if (bets['1st12'] && result >= 1 && result <= 12) winnings += bets['1st12'] * 2;
  // etc...

  alert(`Result: ${result} — You won $${winnings}`);
}
